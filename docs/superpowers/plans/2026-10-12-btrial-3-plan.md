```cpp
#pragma once

#include <algorithm>

// Geometry for BaseTheme::drawHeader and drawHeaderWithAction, free of the
// renderer and FreeInkUI so it can be host-tested
// (test/ui_layout/HeaderLayoutTest.cpp). All coordinates are absolute.
namespace HeaderLayout {

struct Box {
  int x;
  int y;
  int width;
  int height;
};

// FreeInkUI's own trailing header button is the band height less twice this,
// inset this far from the band's top (header.h).
constexpr int ACTION_INSET = 4;
// A detached battery hugs the corner of its strip at the legacy inset.
constexpr int DETACHED_BATTERY_INSET = 12;

// TagChipView::measure's width, reproduced because the theme cannot include
// reader code: the bold label plus spaceLg each side, never narrower than a
// chip is tall (TagChips::chipHeight).
constexpr int actionWidth(const int boldTextWidth, const int textLineHeight, const int spaceLg, const int spaceMd,
                          const int minTouchSize) {
  const int chipHeight = std::max(textLineHeight + 2 * spaceMd, minTouchSize);
  return std::max(boldTextWidth + 2 * spaceLg, chipHeight);
}

struct Inputs {
  Box band{0, 0, 0, 0};
  int sidePadding = 0;
  int underline = 0;
  int spaceMd = 0;
  int titleLineHeight = 0;
  int batteryReserve = 0;
  int batteryBarHeight = 0;
  bool batteryLeft = false;
  bool batteryDetached = false;
  bool titleCentred = false;
  // 0 lays out drawHeader's header, with no action.
  int actionWidth = 0;
};

struct Result {
  int leftReserve = 0;
  int rightReserve = 0;
  int titleOffsetY = 0;
  Box battery{0, 0, 0, 0};
  Box action{0, 0, 0, 0};
};

constexpr Result layout(const Inputs& in) {
  const Box& band = in.band;
  const int bandRight = band.x + band.width;
  Result out;

  if (in.batteryDetached) {
    // The battery has its own corner strip, so the title owns the lower
    // sub-band's full width, anchored with clearance above the bottom rule;
    // naive sub-band centring left the glyphs nearly touching it.
    const int titleTop = band.height - in.underline - in.spaceMd - in.titleLineHeight;
    out.titleOffsetY = titleTop - (band.height - in.titleLineHeight) / 2;
  } else if (in.batteryLeft) {
    out.leftReserve = in.batteryReserve + in.spaceMd;
  } else {
    out.rightReserve = in.batteryReserve + in.spaceMd;
  }

  const int batteryInset = in.batteryDetached ? DETACHED_BATTERY_INSET : in.sidePadding;
  const int batteryX = in.batteryLeft ? band.x + batteryInset : bandRight - batteryInset - in.batteryReserve;
  out.battery = Box{batteryX, band.y, in.batteryReserve, in.batteryBarHeight};

  return out;
}

}  // namespace HeaderLayout
```

```cpp
// "Done" (about 52 px of bold text) and a label narrow enough to hit the floor.
constexpr int DONE_W = 52 + 2 * SPACE_LG;
constexpr int FLOOR_W = MIN_TOUCH;

bool intersects(const Box& a, const Box& b) {
  if (a.width <= 0 || a.height <= 0 || b.width <= 0 || b.height <= 0) return false;
  return a.x < b.x + b.width && b.x < a.x + a.width && a.y < b.y + b.height && b.y < a.y + a.height;
}

// The rect fui::header draws the title into: the band inset by the side
// padding and the reserves, shifted by titleOffsetY.
Box titleBox(const Inputs& in, const Result& r) {
  return Box{in.band.x + in.sidePadding + r.leftReserve, in.band.y + r.titleOffsetY,
             in.band.width - 2 * in.sidePadding - r.leftReserve - r.rightReserve, in.band.height};
}
```

```cpp
#include <algorithm>

// Geometry for BaseTheme::drawHeader and drawHeaderWithAction, free of the
// renderer and FreeInkUI so it can be host-tested
// (test/ui_layout/HeaderLayoutTest.cpp). All coordinates are absolute.
namespace HeaderLayout {

struct Box {
  int x;
  int y;
  int width;
  int height;
};

// FreeInkUI's own trailing header button is the band height less twice this,
// inset this far from the band's top (header.h).
constexpr int ACTION_INSET = 4;
// A detached battery hugs the corner of its strip at the legacy inset.
constexpr int DETACHED_BATTERY_INSET = 12;

// TagChipView::measure's width, reproduced because the theme cannot include
// reader code: the bold label plus spaceLg each side, never narrower than a
// chip is tall (TagChips::chipHeight).
constexpr int actionWidth(const int boldTextWidth, const int textLineHeight, const int spaceLg, const int spaceMd,
                          const int minTouchSize) {
  const int chipHeight = std::max(textLineHeight + 2 * spaceMd, minTouchSize);
  return std::max(boldTextWidth + 2 * spaceLg, chipHeight);
}

struct Inputs {
  Box band{0, 0, 0, 0};
  int sidePadding = 0;
  int underline = 0;
  int spaceMd = 0;
  int titleLineHeight = 0;
  int batteryReserve = 0;
  int batteryBarHeight = 0;
  bool batteryLeft = false;
  bool batteryDetached = false;
  bool titleCentred = false;
  // 0 lays out drawHeader's header, with no action.
  int actionWidth = 0;
};

struct Result {
  int leftReserve = 0;
  int rightReserve = 0;
  int titleOffsetY = 0;
  Box battery{0, 0, 0, 0};
  Box action{0, 0, 0, 0};
};

constexpr Result layout(const Inputs& in) {
  const Box& band = in.band;
  const int bandRight = band.x + band.width;
  Result out;

  if (in.batteryDetached) {
    // The battery has its own corner strip, so the title owns the lower
    // sub-band's full width, anchored with clearance above the bottom rule;
    // naive sub-band centring left the glyphs nearly touching it.
    const int titleTop = band.height - in.underline - in.spaceMd - in.titleLineHeight;
    out.titleOffsetY = titleTop - (band.height - in.titleLineHeight) / 2;
  } else if (in.batteryLeft) {
    out.leftReserve = in.batteryReserve + in.spaceMd;
  } else {
    out.rightReserve = in.batteryReserve + in.spaceMd;
  }

  const int batteryInset = in.batteryDetached ? DETACHED_BATTERY_INSET : in.sidePadding;
  const int batteryX = in.batteryLeft ? band.x + batteryInset : bandRight - batteryInset - in.batteryReserve;
  out.battery = Box{batteryX, band.y, in.batteryReserve, in.batteryBarHeight};

  if (in.actionWidth <= 0) return out;

  const bool batteryRight = !in.batteryLeft;
  const bool sharesTitleLine = batteryRight && !in.batteryDetached;
  const bool corner = sharesTitleLine && in.batteryBarHeight < band.height;
  const bool besideBattery = sharesTitleLine && !corner;
  const int roomBottom = band.y + band.height - in.underline;
  const int roomTop = (corner || (in.batteryDetached && batteryRight)) ? band.y + in.batteryBarHeight : band.y;

  int actionRight = bandRight - in.sidePadding;
  int growth = in.actionWidth + in.spaceMd;
  if (besideBattery) {
    actionRight -= in.batteryReserve + in.spaceMd;
  } else if (corner) {
    // The strip above already reserves the battery's columns; the action
    // shares them at a different height, so the title gives up only the excess.
    growth = std::max(0, in.actionWidth - in.batteryReserve);
  }
  out.rightReserve += growth;
  // Equal growth on both sides keeps a centred title where drawHeader centres it.
  if (in.titleCentred) out.leftReserve += growth;

  const int height = std::max(0, std::min(band.height - 2 * ACTION_INSET, roomBottom - roomTop));
  int y = band.y + ACTION_INSET;
  if (corner) {
    y = roomBottom - height;
  } else if (in.batteryDetached) {
    const int titleLineTop = band.y + band.height - in.underline - in.spaceMd - in.titleLineHeight;
    y = titleLineTop + (in.titleLineHeight - height) / 2;
  }
  y = std::min(std::max(y, roomTop), roomBottom - height);
  out.action = Box{actionRight - in.actionWidth, y, in.actionWidth, height};
  return out;
}

}  // namespace HeaderLayout
```

```cpp
Rect BaseTheme::drawHeaderBand(const GfxRenderer& renderer, Rect rect, const char* title, const char* subtitle,
                               const char* actionLabel) const {
  // Every activity header renders through the FreeInkUI header + battery
  // indicator components, styled by the active theme's tokens (padding,
  // centering, underline). Non-interactive frame: no hit rects registered.
  namespace fui = freeink::ui;
  const auto spec = uiScaleSpec();
  fui::GfxRendererFrame<1> ui(renderer, spec.smallFontId, spec.bodyFontId, spec.titleFontId);
  // Refresh the app-wide shared tokens instead of copying ~1.5KB of
  // ThemeTokens onto this render-path stack frame; the values derived here
  // are identical to what every FreeInkApp screen derives. Goes through the
  // same publish-a-fresh-slot path applySharedUiTheme() uses (see
  // UiAppHelpers.h) rather than overwriting the previously-published
  // instance in place, since some other FreeInkApp could be mid-read of it.
  const fui::ThemeTokens& tokens = refreshSharedUiThemeTokens(ui.target);
  // The caller draws the action as a tag chip in the uiScale small font, bold,
  // so it is measured before the small slot is rebound below.
  int actionWidth = 0;
  if (actionLabel != nullptr && actionLabel[0] != '\0') {
    fui::TextStyle actionText = tokens.smallText;
    actionText.bold = true;
    const int textWidth = ui.target.measureText(actionText.font, actionLabel, actionText).width;
    actionWidth = HeaderLayout::actionWidth(textWidth, ui.target.lineHeight(actionText.font), tokens.spaceLg,
                                            tokens.spaceMd, tokens.minTouchSize);
  }
  // Header status text (battery percent, right label) stays at the fixed
  // small font like the legacy headers; the uiScale small font is for list
  // subtitles.
  ui.target.setFont(fui::GfxRendererTarget::FONT_SMALL, SMALL_FONT_ID);
  const ThemeMetrics& metrics = UITheme::getInstance().getMetrics();
  const fui::Rect band{static_cast<int16_t>(rect.x), static_cast<int16_t>(rect.y), static_cast<int16_t>(rect.width),
                       static_cast<int16_t>(rect.height)};

  const bool showBatteryPercentage =
      SETTINGS.hideBatteryPercentage != CrossPointSettings::HIDE_BATTERY_PERCENTAGE::HIDE_ALWAYS;
  const uint16_t percentage = powerManager.getBatteryPercentage();
  char percentText[8];
  snprintf(percentText, sizeof(percentText), "%u%%", static_cast<unsigned>(percentage));
  // The icon glyph extends 2px past glyphWidth (terminal nub); reserve it or
  // the percent label's rect comes up short and the text truncates.
  constexpr int16_t batteryNubWidth = 2;
  int16_t batteryReserve = static_cast<int16_t>(metrics.batteryWidth + batteryNubWidth);
  if (showBatteryPercentage) {
    batteryReserve = static_cast<int16_t>(
        batteryReserve + batteryPercentSpacing +
        ui.target.measureText(fui::GfxRendererTarget::FONT_SMALL, percentText, tokens.smallText).width);
  }

  fui::HeaderProps props;
  props.title = title;
  props.rightLabel = subtitle;  // firmware headers right-align the secondary text
  const bool batteryLeft = metrics.headerBatterySide == 1;
  const bool batteryDetached = metrics.headerBatteryDetached;
  // Shared-line headers with the battery on the right: the header component
  // places rightLabel beside the battery reserve, so it shifts with the
  // percent label's width. While the battery sits in a top strip shorter than
  // the band, draw the label manually below instead, pinned in the band's
  // lower-right corner, so it holds one position across battery states. A
  // strip as tall as the band fills that corner, so the label stays with the
  // component on the title line.
  const bool batteryLeavesCornerFree = metrics.batteryBarHeight < rect.height;
  const bool manualRightLabel = subtitle != nullptr && !batteryDetached && !batteryLeft && batteryLeavesCornerFree;
  if (manualRightLabel) {
    props.rightLabel = nullptr;
  }
  props.borderEdges = fui::EdgeBottom;
  props.titleText = tokens.titleText;
  props.titleText.align = tokens.headerTitleAlign;
  props.subtitleText = tokens.smallText;
  props.styles = tokens.popup;
  props.sidePadding = tokens.headerSidePadding;

  HeaderLayout::Inputs layoutInputs;
  layoutInputs.band = HeaderLayout::Box{rect.x, rect.y, rect.width, rect.height};
  layoutInputs.sidePadding = tokens.headerSidePadding;
  layoutInputs.underline = tokens.headerUnderline;
  layoutInputs.spaceMd = tokens.spaceMd;
  layoutInputs.titleLineHeight = ui.target.lineHeight(fui::GfxRendererTarget::FONT_TITLE);
  layoutInputs.batteryReserve = batteryReserve;
  layoutInputs.batteryBarHeight = metrics.batteryBarHeight;
  layoutInputs.batteryLeft = batteryLeft;
  layoutInputs.batteryDetached = batteryDetached;
  layoutInputs.titleCentred = tokens.headerTitleAlign == fui::TextAlign::Center;
  layoutInputs.actionWidth = actionWidth;
  const HeaderLayout::Result placed = HeaderLayout::layout(layoutInputs);
  props.leftReserve = static_cast<int16_t>(placed.leftReserve);
  props.rightReserve = static_cast<int16_t>(placed.rightReserve);
  props.titleOffsetY = static_cast<int16_t>(placed.titleOffsetY);

  // Underline only under a titled header: an untitled band (Lyra home screen)
  // historically drew no rule, and the old themes keyed the line on the title.
  if (title != nullptr && props.styles.normal.border.kind == fui::PaintKind::None && tokens.headerUnderline > 0) {
    props.styles.normal.border = fui::Paint::solid(fui::Color::Black);
    props.styles.normal.borderWidth = tokens.headerUnderline;
  }
  fui::header(ui.frame, band, props);

  fui::BatteryIndicatorProps battery;
  battery.percent = static_cast<uint8_t>(percentage > 100 ? 100 : percentage);
  battery.charging = gpio.isUsbConnected();
  battery.label = showBatteryPercentage ? percentText : nullptr;
  battery.text = tokens.smallText;
  battery.glyphWidth = static_cast<int16_t>(metrics.batteryWidth);
  battery.glyphHeight = static_cast<int16_t>(metrics.batteryHeight);
  battery.gap = batteryPercentSpacing;
  fui::batteryIndicator(ui.frame,
                        fui::Rect{static_cast<int16_t>(placed.battery.x), static_cast<int16_t>(placed.battery.y),
                                  static_cast<int16_t>(placed.battery.width),
                                  static_cast<int16_t>(placed.battery.height)},
                        battery);

  if (manualRightLabel) {
    const fui::Size labelSize = ui.target.measureText(fui::GfxRendererTarget::FONT_SMALL, subtitle, tokens.smallText);
    const int16_t labelH = ui.target.lineHeight(fui::GfxRendererTarget::FONT_SMALL);
    const fui::Rect labelRect{static_cast<int16_t>(band.right() - tokens.headerSidePadding - labelSize.width),
                              static_cast<int16_t>(band.bottom() - tokens.headerUnderline - tokens.spaceSm - labelH),
                              labelSize.width, labelH};
    ui.target.text(labelRect, subtitle, tokens.smallText);
  }
  return Rect{placed.action.x, placed.action.y, placed.action.width, placed.action.height};
}
```
