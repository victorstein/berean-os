**Step 0.1 — the anchored-cut helper.** Write this file (outside the repo):

```sh
cat > "$TMPDIR/cut.py" <<'EOF'
#!/usr/bin/env python3
"""cut.py FILE START END — delete from the line that starts with START up to (not including)
the line that starts with END. END may be the literal EOF. Fails loudly unless both anchors match
exactly once, so a drifted file stops the plan instead of cutting the wrong span."""
import sys
path, start, end = sys.argv[1], sys.argv[2], sys.argv[3]
lines = open(path, encoding="utf-8").read().split("\n")
s = [i for i, l in enumerate(lines) if l.startswith(start)]
if len(s) != 1:
    sys.exit(f"cut.py: start anchor matched {len(s)} times in {path}: {start!r}")
if end == "EOF":
    e = len(lines)
    while e > s[0] and lines[e - 1] == "":
        e -= 1
    del lines[s[0]:e]
    while lines and lines[-1] == "":
        lines.pop()
    lines.append("")
else:
    es = [i for i, l in enumerate(lines) if l.startswith(end) and i > s[0]]
    if len(es) != 1:
        sys.exit(f"cut.py: end anchor matched {len(es)} times after start in {path}: {end!r}")
    del lines[s[0]:es[0]]
open(path, "w", encoding="utf-8").write("\n".join(lines))
print(f"cut.py: {path}: removed lines {s[0]+1}..")
EOF
```

python3 - <<'EOF'
import glob, re
pat = re.compile(r'^(STR_CALIBRE_[A-Z0-9_]+|STR_ALREADY_ASSIGNED|STR_BROWSE_FILES|STR_HW_(BACK|CONFIRM|LEFT|RIGHT)_LABEL'
                 r'|STR_MENU_RECENT_BOOKS|STR_NO_RECENT_BOOKS|STR_REMAP_(CANCEL_HINT|PROMPT|RESET_HINT|FRONT_BUTTONS)'

                 r'|STR_NO_OPEN_BOOK|STR_START_READING):')
total = 0
for path in sorted(glob.glob("lib/I18n/translations/*.yaml")):
    with open(path, encoding="utf-8") as f:
        lines = f.readlines()
    kept = [l for l in lines if not pat.match(l)]

print("removed", total, "lines")
EOF
