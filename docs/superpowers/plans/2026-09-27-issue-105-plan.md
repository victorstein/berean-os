## How this plan is built, read before starting

- **All page edits are scripted and pinned.** `FilesPage.html` is 6,036 lines. The strip is 14
  line-range edits to the markup and JS, 6 to the CSS, and one to the `<script src>` tag, all
  addressed by **e41880e3 baseline line numbers**. `strip_filespage.py` (below) applies them in
  three stages. Each stage refuses to run unless the page's sha256 equals that stage's expected
  input, so it cannot apply to a page that has drifted, whether through a sibling's edit or a
  half-run stage. Each stage's edits all sit *below* every later stage's edits, so baseline
  numbering stays valid from stage to stage.
- **The test is `check_filespage.py`** (spec A8 and its checks 1-7). It is **not committed**: the
  repo has no HTML/JS harness, and adding one would be a new pattern. It lives in a scratch
  directory.
- **Both scripts were run end to end on a copy of the baseline page while this plan was written.**
  The expected output of every run below is the real output from that trial, so a deviation means
  something changed.
- **Code style.** Per `CLAUDE.md` "Comments" and the user's global rule: comments describe the
  merged state, and only non-obvious *why* earns one. The replacement blocks below follow that
  rule; keep them verbatim.

```python
#!/usr/bin/env python3
"""Static check for issue #105: FilesPage.html carries no optimiser and no dangling reference.

Usage: check_filespage.py <repo-root> [page-path]
Compares the page against the pinned baseline at e41880e3 and exits 1 on any failure.
"""
import re
import subprocess
import sys
import tempfile
from pathlib import Path

BASELINE_REV = "e41880e3"
PAGE_REL = "src/network/html/FilesPage.html"

# Baseline line ranges (1-based, inclusive) that the spec removes from the <script> block.
REMOVED_JS_RANGES = [(2207, 2968), (3123, 3141), (3283, 3942), (3975, 5311)]
# The rest of the baseline <script> block: names declared here are kept even if a removed range reuses them.
KEPT_JS_RANGES = [(1843, 2206), (2969, 3122), (3142, 3282), (3943, 3974), (5312, 6034)]
# Baseline markup ranges the spec removes.
REMOVED_MARKUP_RANGES = [(1565, 1751), (1758, 1763)]

JS_KEYWORDS = {
    "if", "for", "while", "switch", "catch", "function", "return", "typeof", "new", "await",
    "async", "super", "import", "delete", "void", "in", "of", "do", "else", "try", "throw",
}


def read_baseline(repo):
    return subprocess.run(
        ["git", "-C", str(repo), "show", f"{BASELINE_REV}:{PAGE_REL}"],
        check=True, capture_output=True, text=True,
    ).stdout


def split_page(text):
    style = re.search(r"<style>(.*?)</style>", text, re.S).group(1)
    script = re.search(r"<script>(.*?)</script>", text, re.S).group(1)
    body = re.search(r"<body>(.*?)<script>", text, re.S).group(1)
    return style, body, script


def strip_js_comments(js):
    js = re.sub(r"/\*.*?\*/", "", js, flags=re.S)
    return re.sub(r"(^|\s)//.*$", r"\1", js, flags=re.M)


def lines_in(text, ranges):
    lines = text.split("\n")
    return "\n".join("\n".join(lines[a - 1:b]) for a, b in ranges)


def top_level_names(text, ranges):
    """Functions, plus let/const/var at the script's top level: indent 2 before the
    3283 banner, indent 0 after it (the baseline script changes indentation there)."""
    lines = text.split("\n")
    names = set()
    for a, b in ranges:
        for n in range(a, b + 1):
            line = lines[n - 1]
            top_indent = "  " if n < 3283 else ""
            names |= set(re.findall(r"\bfunction\s+([A-Za-z_$][\w$]*)", line))
            m = re.match(rf"^{top_indent}(?:let|const|var)\s+([A-Za-z_$][\w$]*)", line)
            if m:
                names.add(m.group(1))
    return names - JS_KEYWORDS


def all_declared(js):
    names = set(re.findall(r"\bfunction\s+([A-Za-z_$][\w$]*)", js))
    names |= set(re.findall(r"\b(?:let|const|var)\s+([A-Za-z_$][\w$]*)", js))
    return names


def removed_ids(markup):
    return set(re.findall(r'\bid="([^"]+)"', markup))


def referenced_ids(script):
    ids = set(re.findall(r"getElementById\('([^']+)'\)", script))
    ids |= set(re.findall(r"querySelector(?:All)?\('#([\w-]+)", script))
    return ids


def markup_ids(body):
    return set(re.findall(r'\bid="([^"]+)"', body))


def inline_handler_calls(body):
    calls = set()
    for code in re.findall(r'\bon\w+="([^"]*)"', body):
        calls |= set(re.findall(r"(?<![.\w$])([A-Za-z_$][\w$]*)\s*\(", code))
    return calls - JS_KEYWORDS


def bare_calls(script):
    return set(re.findall(r"(?<![.\w$])([A-Za-z_$][\w$]*)\s*\(", script)) - JS_KEYWORDS


def used_classes(body, script):
    classes = set()
    for attr in re.findall(r'class="([^"]*)"', body + script):
        classes |= {c for c in attr.split() if re.fullmatch(r"[\w-]+", c)}
    classes |= set(re.findall(r"classList\.(?:add|remove|toggle|contains)\('([\w-]+)'", script))
    classes |= set(re.findall(r"\.closest\('\.([\w-]+)'\)", script))
    return classes


def styled_classes(style):
    return set(re.findall(r"\.([A-Za-z_][\w-]*)", re.sub(r"\{[^{}]*\}", "{}", style)))


def run(repo, page_path):
    baseline = read_baseline(repo)
    page = Path(page_path).read_text(encoding="utf-8")
    b_style, b_body, b_script = split_page(baseline)
    style, body, script = split_page(page)
    failures = []

    # 1. No jszip, no external script.
    if re.search(r"jszip", page, re.I) or "<script src=" in page:
        failures.append("1: page still loads or names jszip")

    # 2. No identifier or id that the spec removes.
    removed_names = top_level_names(baseline, REMOVED_JS_RANGES)
    removed_names -= all_declared(lines_in(baseline, KEPT_JS_RANGES))
    code = strip_js_comments(script)
    leftovers = sorted(n for n in removed_names if re.search(rf"(?<![.\w$]){re.escape(n)}\b", code))
    ids_gone = removed_ids(lines_in(baseline, REMOVED_MARKUP_RANGES))
    leftover_ids = sorted(i for i in ids_gone if re.search(rf"['\"#]{re.escape(i)}['\"]", script + body))
    if leftovers:
        failures.append(f"2: removed identifiers still referenced: {leftovers}")
    if leftover_ids:
        failures.append(f"2: removed element ids still referenced: {leftover_ids}")

    # 3. Every id the script looks up exists, beyond what the baseline already created at runtime.
    baseline_dynamic_ids = referenced_ids(b_script) - markup_ids(b_body)
    missing_ids = sorted(referenced_ids(script) - markup_ids(body) - baseline_dynamic_ids)
    if missing_ids:
        failures.append(f"3: script looks up ids absent from markup: {missing_ids}")

    # 4. Every inline handler calls a defined function.
    b_globals = inline_handler_calls(b_body) - all_declared(b_script)
    missing_handlers = sorted(inline_handler_calls(body) - all_declared(script) - b_globals)
    if missing_handlers:
        failures.append(f"4: inline handlers call undefined functions: {missing_handlers}")

    # 5. No class used without a style rule, beyond the baseline's own JS-only hooks.
    b_unstyled = used_classes(b_body, b_script) - styled_classes(b_style)
    unstyled = sorted(used_classes(body, script) - styled_classes(style) - b_unstyled)
    if unstyled:
        failures.append(f"5: classes used with no CSS rule: {unstyled}")

    # 6. The script parses.
    with tempfile.NamedTemporaryFile("w", suffix=".js", delete=False) as tmp:
        tmp.write(script)
    node = subprocess.run(["node", "--check", tmp.name], capture_output=True, text=True)
    if node.returncode != 0:
        failures.append("6: node --check failed:\n" + node.stderr)

    # 7. Every bare call resolves, beyond the baseline's own globals and callback parameters.
    b_unresolved = bare_calls(b_script) - all_declared(b_script)
    unresolved = sorted(bare_calls(script) - all_declared(script) - b_unresolved)
    if unresolved:
        failures.append(f"7: calls to undefined names: {unresolved}")

    print(f"removed-name list: {len(removed_names)} names, {len(ids_gone)} ids")
    for f in failures:
        print("FAIL", f)
    print("PASS" if not failures else f"{len(failures)} check(s) failed")
    return 1 if failures else 0


if __name__ == "__main__":
    repo = Path(sys.argv[1])
    sys.exit(run(repo, sys.argv[2] if len(sys.argv) > 2 else repo / PAGE_REL))
```
